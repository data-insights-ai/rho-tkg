out = []
for m in mutants:
    name, path, old, new, runs = m[:5]
    nth = m[5] if len(m) > 5 else None
    src = open(path).read()
    if nth is None and src.count(old) != 1:
        out.append('%s: PATTERN NOT FOUND (%d)' % (name, src.count(old)))
        continue
    if nth is None:
        mutated = src.replace(old, new)
    else:
        idx = -1
        for _ in range(nth):
            idx = src.index(old, idx + 1)
        mutated = src[:idx] + new + src[idx + len(old):]
    open(path, 'w').write(mutated)
    try:
        killed_by = None
        for pkg, run in runs:
            r = subprocess.run(['go', 'test', '-count=1', '-run', run, pkg], capture_output=True, text=True, timeout=600)
            if r.returncode != 0:
                first = next((l.strip() for l in r.stdout.splitlines() if l.strip().startswith(('valid_end', 'effective_validfrom', 'badgerstore_valid_end', 'memorystore_valid_end')) or 'FAIL:' in l), r.stdout.strip().splitlines()[-1] if r.stdout.strip() else r.stderr.strip()[:200])
                killed_by = '%s -run %s: %s' % (pkg, run, first[:220])
                break
        out.append('%s: %s' % (name, ('KILLED by ' + killed_by) if killed_by else 'SURVIVED'))
    finally:
        open(path, 'w').write(src)
    print(out[-1], flush=True)

open('tasks/evidence/effective-validfrom/06-mutants.txt', 'w').write('# Break-the-code mutants (each applied alone, targeted tests run, source restored)\n' + '\n'.join(out) + '\n')

package core

import (
	"testing"

	"github.com/data-insights-ai/rho-tkg/v4/pkg/types"
)

// Repro (backlog 34): a backfilled re-import whose tkg_tx_from lies inside
// the first life (before its delete D) joins the first life in the chain's
// write order (life = deletes recorded before the row's TxFrom), so lifeEnds
// caps it at D: the as-of doors read it present at a pin after the import,
// the valid-time doors absent at every valid instant >= D.
func TestZZBackfilledReImportBeforeDelete(t *testing.T) {
	ccRun(t, true, func(t *testing.T, mk func(t *testing.T) *ccEnt) {
		e := mk(t)
		b := e.add("B", 1000, nil)
		id := e.add("T", 1000, nil)
		backfill := e.pin()
		e.mustUpdate(id, map[string]any{"x": int64(1)})
		e.mustDel(id)
		d := e.deletedAt(id)
		props := map[string]any{"tkg_valid_from": types.Instant(7000), "x": int64(9), "tkg_tx_from": backfill}
		var curV uint32
		if e.rel {
			s, _ := e.g.Nodes.Get(e.ctx, e.start)
			en, _ := e.g.Nodes.Get(e.ctx, e.end)
			r, err := e.g.Rels.Import(e.ctx, types.RelID(id), ccType, s, en, props)
			if err != nil {
				t.Fatalf("Rels.Import: %v", err)
			}
			curV = r.Version()
		} else {
			n, err := e.g.Nodes.Import(e.ctx, types.NodeID(id), []string{ccLabel}, props)
			if err != nil {
				t.Fatalf("Nodes.Import: %v", err)
			}
			curV = n.Version()
		}
		now := e.pin()
		want := ccSet(ccVer("B", 0), ccVer("T", curV))
		e.expect("as-of now", e.asOfDoors(now), want, id, b)
		e.expect("valid D+1 now", e.atTxDoors(d+1, now), want, id, b)
		e.expect("valid D+1", e.validDoors(d+1), want, id, b)
	})
}

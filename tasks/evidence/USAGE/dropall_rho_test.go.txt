package badger

import (
	"fmt"
	"strings"
	"testing"

	snowflake "github.com/bds421/rho-snowflake-2026"
	"github.com/data-insights-ai/rho-tkg/v4/pkg/types"
)

// DROPALL at rho level: GetNode of a 1.4 MB row (value log) has found the key
// when Clear (store without the change log: DropAll) runs; then it takes the value.
func TestUsageDropAllMidRead(t *testing.T) {
	for _, size := range []int{100_000, 1_400_000} {
		t.Run(fmt.Sprint(size), func(t *testing.T) {
			dir := t.TempDir()
			bs, err := New(Config{Dir: dir})
			if err != nil {
				t.Fatal(err)
			}
			n := types.NewNode(types.NodeID(snowflake.ID(77)), 7, nil)
			for i := 0; i*60_000 < size; i++ {
				_ = n.SetProperty(fmt.Sprintf("p%02d", i), strings.Repeat("v", 60_000))
			}
			if err := bs.PutNode(n); err != nil {
				t.Fatal(err)
			}
			if err := bs.Close(); err != nil {
				t.Fatal(err)
			}
			bs, err = New(Config{Dir: dir}) // a fresh cache: GetNode reads badger
			if err != nil {
				t.Fatal(err)
			}
			defer bs.Close()
			paused, resume := make(chan struct{}), make(chan struct{})
			zzGetNodeAfterGetHook = func() { close(paused); <-resume }
			defer func() { zzGetNodeAfterGetHook = nil }()
			type res struct {
				n   *types.Node
				err error
			}
			out := make(chan res, 1)
			go func() { got, err := bs.GetNode(n.ID()); out <- res{got, err} }()
			<-paused
			zzGetNodeAfterGetHook = nil
			clearErr := bs.Clear()
			close(resume)
			r := <-out
			t.Logf("DROPALL row of %d B: Clear err %v; the read that had found the key returns: node %v, err %.120v", size, clearErr, r.n != nil, r.err)
		})
	}
}
